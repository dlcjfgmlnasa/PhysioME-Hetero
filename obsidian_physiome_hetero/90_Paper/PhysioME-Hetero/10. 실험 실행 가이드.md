## STEP 4 — Downstream evaluation

```bash
CKPT=ckpt/vital_db_hetero/model/best_model.pth
DATA=data/vitaldb_downstream

# IOH primary task
python downstream/run_ioh.py \
    --ckpt_path $CKPT \
    --data_dir  $DATA \
    --out_dir   results \
    --tag       hetero

# IOH A1 baseline
python downstream/run_ioh.py \
    --ckpt_path ckpt/vital_db/physiome/model/best_model.pth \
    --data_dir  $DATA \
    --out_dir   results \
    --tag       synth_only

# AKI secondary task
python downstream/run_aki.py \
    --ckpt_path    $CKPT \
    --data_dir     $DATA \
    --clinical_csv data/vitaldb_clinical.csv \
    --lab_csv      data/vitaldb_lab.csv \
    --out_dir      results \
    --tag          hetero

# External transfer (MIMIC-III mortality)
python downstream/run_mortality.py \
    --ckpt_path    $CKPT \
    --mimic_npz_dir data/mimic3_hetero \
    --cohort_csv    data/icu_mortality_cohort.csv \
    --mode         zero_shot \
    --out_dir      results \
    --tag          hetero
```
